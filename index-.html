// 1. URL de tu Webhook en Make const URL_MAKE = "https://hook.us2.make.com/7nfgc21f08o4a2uqxvqbt06aqoa119kv"; // 2. Función que se ejecuta al detectar un QR function onScanSuccess(decodedText) { console.log("QR Detectado:", decodedText); document.getElementById('resultado').value = decodedText; document.getElementById('msg').innerText = "¡QR Detectado!"; document.getElementById('msg').style.color = "green"; } // 3. Función para enviar los datos (Asegúrate de que el 'return' esté aquí dentro) function enviar() { const dato = document.getElementById('resultado').value; if (!dato) { alert("Escanea un QR primero"); return; // Aquí el return es legal porque está dentro de la función enviar } const msgElement = document.getElementById('msg'); msgElement.innerText = "Enviando..."; msgElement.style.color = "orange"; fetch(URL_MAKE + "?qr=" + encodeURIComponent(dato), { method: 'GET', mode: 'no-cors' // Esto ayuda si tienes problemas de CORS con Make }) .then(() => { alert("¡Dato enviado con éxito!"); msgElement.innerText = "Enviado correctamente"; msgElement.style.color = "green"; }) .catch(err => { alert("Hubo un error al enviar."); console.error(err); }); } // 4. Inicializar el escáner (Esto debe ir al final) let scanner = new Html5QrcodeScanner("reader", { fps: 10, qrbox: {width: 250, height: 250} }); scanner.render(onScanSuccess);
